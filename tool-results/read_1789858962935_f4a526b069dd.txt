   428→  // ── توابع داخلی ──
   429→  function toggleGood(goodId: string) {
   430→    setDraft((d) => {
   431→      const next = { ...d };
   432→      if (next[goodId]) {
   433→        delete next[goodId];
   434→      } else {
   435→        next[goodId] = {
   436→          mode: "BOTH",
   437→          sell: { price: 0, stock: 0, minOrder: 0 },
   438→          buy: { volume: 0, frequency: "MONTHLY" },
   439→        };
   440→      }
   441→      return next;
   442→    });
   443→  }
   444→
   445→  function patchMode(goodId: string, mode: Mode) {
   446→    setDraft((d) => ({ ...d, [goodId]: { ...d[goodId], mode } }));
   447→  }
   448→
   449→  function patchDraft<K extends "sell" | "buy">(
   450→    goodId: string,
   451→    part: K,
   452→    patch: Partial<DraftItem[K]>
   453→  ) {
   454→    setDraft((d) => ({
   455→      ...d,
   456→      [goodId]: { ...d[goodId], [part]: { ...d[goodId][part], ...patch } },
   457→    }));
   458→  }
   459→
   460→  async function submit() {
   461→    if (!biz) return;
   462→    if (selectedIds.length === 0) {
   463→      toast({ title: "حداقل یک کالا انتخاب کنید", variant: "destructive" });
   464→      return;
   465→    }
   466→    for (const id of selectedIds) {
   467→      const item = draft[id];
   468→      const g = goodById.get(id);
   469→      if (!g || !item) continue;
   470→      if (item.mode !== "BUY" && (item.sell.price <= 0 || item.sell.stock <= 0)) {
   471→        setErrorGood(id);
   472→        toast({
   473→          title: `مشخصات فروش «${g.name}» کامل نیست`,
   474→          description: "قیمت و موجودی فروش را وارد کنید.",
   475→          variant: "destructive",
   476→        });
   477→        return;
   478→      }
   479→      if (item.mode !== "SELL" && item.buy.volume <= 0) {
   480→        setErrorGood(id);
   481→        toast({
   482→          title: `مشخصات خرید «${g.name}» کامل نیست`,
   483→          description: "حجم خرید در هر دوره را وارد کنید.",
   484→          variant: "destructive",
   485→        });
   486→        return;
   487→      }
   488→    }
   489→    setErrorGood(null);
   490→
   491→    const upsert = upsertMutation;
   492→    try {
   493→      for (const id of selectedIds) {
   494→        const item = draft[id];
   495→        await upsert.mutateAsync({
   496→          businessId: biz.id,
   497→          goodId: id,
   498→          mode: item.mode,
   499→          ...(item.mode !== "BUY" ? { sell: item.sell } : {}),
   500→          ...(item.mode !== "SELL" ? { buy: item.buy } : {}),
   501→        });
   502→      }
   503→      toast({ title: "بازوها ساخته شد", description: "دو لینک اختصاصی برای شما آماده است." });
   504→      setStep("done");
   505→    } catch (err) {
   506→      toast({
   507→        title: "ثبت کالاها ناموفق بود",
   508→        description: err instanceof ApiError ? err.message : "دوباره تلاش کنید",
   509→        variant: "destructive",
   510→      });
   511→    }
   512→  }
   513→}

   528→
   529→  const submit = async () => {
